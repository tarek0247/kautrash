const ADDRESS_API =
  "https://boundaries.biip.lt/v1";

/*
  The API uses cursor pagination.
  We keep requesting pages until next_page is empty.
*/

async function fetchAllPages(
  endpoint,
  filters = [],
  options = {}
) {
  const allItems = [];
  let cursor = null;

  const size =
    options.size || 100;

  let safety = 0;

  while (safety < 100) {
    safety += 1;

    const query = new URLSearchParams();

    query.append(
      "size",
      String(size)
    );

    query.append(
      "sort_by",
      "name"
    );

    query.append(
      "sort_order",
      "asc"
    );

    if (cursor) {
      query.append(
        "cursor",
        cursor
      );
    }

    const response = await fetch(
      `${ADDRESS_API}${endpoint}?${query.toString()}`,
      {
        method: "POST",
        headers: {
          Accept: "application/json",
          "Content-Type":
            "application/json",
        },
        body: JSON.stringify({
          filters,
        }),
      }
    );

    if (!response.ok) {
      const body =
        await safeResponseText(response);

      throw new Error(
        `Address API HTTP ${response.status}${
          body ? `: ${body}` : ""
        }`
      );
    }

    const json =
      await response.json();

    const items =
      Array.isArray(json?.items)
        ? json.items
        : [];

    allItems.push(...items);

    const next =
      json?.next_page ||
      json?.next_cursor ||
      null;

    if (!next || items.length === 0) {
      break;
    }

    cursor = String(next);
  }

  return allItems;
}

async function safeResponseText(
  response
) {
  try {
    return await response.text();
  } catch {
    return "";
  }
}

  /* =======================================================
     FIND MY LOCATION
     ======================================================= */

  async function findMyLocation() {
    try {
      setLocationLoading(true);

      const permission =
        await Location.requestForegroundPermissionsAsync();

      if (
        permission.status !==
        "granted"
      ) {
        Alert.alert(
          "Location permission",
          "Please allow location access in your phone settings."
        );

        return;
      }

      const enabled =
        await Location.hasServicesEnabledAsync();

      if (!enabled) {
        Alert.alert(
          "Location is off",
          "Turn on Location Services on your phone and try again."
        );

        return;
      }

      const position =
        await Location.getCurrentPositionAsync(
          {
            accuracy:
              Location.Accuracy.High,
          }
        );

      const latitude =
        position.coords.latitude;

      const longitude =
        position.coords.longitude;

      console.log(
        "GPS:",
        latitude,
        longitude
      );

      let results = [];

      try {
        results =
          await Location.reverseGeocodeAsync(
            {
              latitude,
              longitude,
            }
          );
      } catch (geocodeError) {
        console.log(
          "Expo reverse geocode error:",
          geocodeError
        );
      }

      if (
        !results ||
        results.length === 0
      ) {
        Alert.alert(
          "Location found",
          "GPS worked, but your phone did not return a street address. Please choose the address manually."
        );

        return;
      }

      const result =
        results[0];

      console.log(
        "Reverse geocode:",
        result
      );

      const detectedCity =
        result.city ||
        result.subregion ||
        result.district ||
        "";

      const detectedStreet =
        result.street ||
        "";

      const detectedNumber =
        result.streetNumber ||
        "";

      const detectedDistrict =
        result.district ||
        "";

      if (detectedCity) {
        setCity(
          detectedCity
        );
      }

      if (detectedStreet) {
        setStreet(
          detectedStreet
        );
      }

      if (detectedNumber) {
        setHouseNumber(
          detectedNumber
        );
      }

      if (detectedDistrict) {
        setSubDistrict(
          detectedDistrict
        );
      }

      try {
        if (
          districtOptions.length === 0
        ) {
          await loadDistricts();
        }
      } catch {
        // Keep GPS result.
      }

      Alert.alert(
        "Location found",
        [
          detectedStreet,
          detectedNumber,
          detectedCity,
        ]
          .filter(Boolean)
          .join(" ")
          ||
            "Coordinates found. Please complete your address."
      );
    } catch (error) {
      console.log(
        "Location error:",
        error
      );

      Alert.alert(
        "Location error",
        error?.message ||
          "Could not determine your location."
      );
    } finally {
      setLocationLoading(false);
    }
  }

  async function findSchedule() {
    const cleanDistrict =
      district.trim();

      subDistrict.trim();

    const cleanCity =
      city.trim();

    const cleanStreet =
      street.trim();

    const cleanHouseNumber =
      houseNumber.trim();

      !cleanDistrict ||
      !cleanSubDistrict ||
      !cleanCity ||
      !cleanStreet ||
      !cleanHouseNumber
    ) {
      Alert.alert(
        "Missing information",
        "Please select the district, seniūnija, city/town, street and enter the house number."
      );

      return;
    }

    const newAddress = {
      district:
        cleanDistrict,

      subDistrict:
        cleanSubDistrict,

      city:
        cleanCity,

      street:
        cleanStreet,

      houseNumber:
        cleanHouseNumber,
    };

    setLoading(true);

    try {
      console.log(
        "Searching Švara:",
        newAddress
      );

      const contracts =
        await getContracts(
          newAddress
        );

      console.log(
        "Švara contracts:",
        contracts
      );

      if (
        !contracts ||
        contracts.length === 0
      ) {
        Alert.alert(
          "No schedule found",
          "Švara did not return a collection schedule for this address."
        );

        return;
      }

      const converted =
        contracts.map(
          (item, index) => {
            const description =
              item.description ||
              item.wasteType ||
              item.type ||
              "";

            return {
              id: String(
                item.id ||
                  item.wasteObjectId ||
                  item.scheduleId ||
                  item.dateFmt ||
                  index
              ),

              title:
                translateWasteType(
                  description
                ),

              wasteType:
                getWasteTypeId(
                  description
                ),

              frequency:
                item.frequency ||
                item.weekDay ||
                item.dateFmt ||
                "Collection scheduled",

              date:
                item.date,

              dateFmt:
                item.dateFmt,

              year:
                item.year,

              month:
                item.month,

              day:
                item.day,

              weekDay:
                item.weekDay,

              street:
                item.street ||
                newAddress.street,

              address:
                item.fullAddress ||
                `${newAddress.street} ${newAddress.houseNumber}`,

              raw:
                item,
            };
          }
        );

      setCurrentAddress(
        newAddress
      );

      setCurrentCollections(
        converted
      );

      /*
        CLOSE ADDRESS SCREEN
        AND GO DIRECTLY HOME.
      */

      setShowAddress(false);

      setPage("home");
    } catch (error) {
      console.log(
        "Švara error:",
        error
      );

      Alert.alert(
        "Connection error",
        error?.message ||
          "Could not connect to Švara."

function PickerModal({
  visible,
  type,
  search,
  setSearch,
  onClose,
  districts,
  subDistricts,
  cities,
  streets,
  onDistrict,
  onSubDistrict,
  onCity,
  onStreet,
  loading,
}) {
  let data = [];
  let title = "";

  if (type === "district") {
    data = districts;
    title = "Choose district";
  }

  if (type === "subDistrict") {
    data = subDistricts;
    title = "Choose seniūnija";
  }

  if (type === "city") {
    data = cities;
    title = "Choose city or town";
  }

  if (type === "street") {
    data = streets;
    title = "Choose street";
  }

  function select(item) {
    if (type === "district") {
      onDistrict(item);
    }

    if (type === "subDistrict") {
      onSubDistrict(item);
    }

    if (type === "city") {
      onCity(item);
    }

    if (type === "street") {
      onStreet(item);
    }
  }

  return (
    <Modal
      visible={visible}
      animationType="slide"
      transparent
      onRequestClose={
        onClose
      }
    >
      <View
        style={
          styles.modalOverlay
        }
      >
        <View
          style={
            styles.modalCard
          }
        >
          <View
            style={
              styles.modalHeader
            }
          >
            <Text
              style={
                styles.modalTitle
              }
            >
              {title}
            </Text>

            <TouchableOpacity
              onPress={onClose}
            >
              <Text
                style={
                  styles.closeText
                }
              >
                ✕
              </Text>
            </TouchableOpacity>
          </View>

          <TextInput
            style={
              styles.searchInput
            }
            placeholder={
              type === "street"
                ? "Search street..."
                : "Search..."
            }
            value={search}
            onChangeText={
              setSearch
            }
            autoFocus={
              type === "street"
            }
            autoCapitalize="none"
          />

          {loading ? (
            <View
              style={
                styles.pickerLoading
              }
            >
              <ActivityIndicator
                color="#4F6548"
              />

              <Text
                style={
                  styles.loadingText
                }
              >
                Loading official address data...
              </Text>
            </View>
          ) : null}

          {!loading &&
          data.length === 0 ? (
            <View
              style={
                styles.noResults
              }
            >
              <Text
                style={
                  styles.noResultsEmoji
                }
              >
                📍
              </Text>

              <Text
                style={
                  styles.noResultsTitle
                }
              >
                No options found
              </Text>

              <Text
                style={
                  styles.noResultsText
                }
              >
                Try another search or choose
                the previous location first.
              </Text>
            </View>
          ) : null}

          <FlatList
            data={data}
            keyExtractor={(
              item,
              index
            ) =>
              String(
                item.code ||
                  item.name ||
                  index
              )
            }
            keyboardShouldPersistTaps="handled"
            renderItem={({
              item,
            }) => (
              <TouchableOpacity
                style={
                  styles.option
                }
                onPress={() =>
                  select(item)
                }
              >
                <Text
                  style={
                    styles.optionText
                  }
                >
                  {item.name}
                </Text>

                <Text
                  style={
                    styles.optionArrow
                  }
                >
                  ›

              </TouchableOpacity>

        </View>
      </View>
    </Modal>
  );
}

function removeDuplicates(
  items
) {
  const map = new Map();

  items.forEach((item) => {
    const key =
      item.code != null
        ? String(item.code)
        : String(
            item.name || ""
          ).toLowerCase();

    if (!map.has(key)) {
      map.set(
        key,
        item
      );
    }
  });

  return Array.from(
    map.values()
  );
}

function filterOptions(
  items,
  search
) {
  if (!search?.trim()) {
    return items;
  }

  const query =
    search
      .toLowerCase()
      .trim();

  return items.filter(
    (item) =>
      String(
        item.name || ""
      )
        .toLowerCase()
        .includes(query)
  );
}

function getWasteTypeId(
  description
) {
  if (!description) {
    return "mixed";
  }

  const value =
    String(
      description
    ).toLowerCase();

  if (
    value.includes("stiklo")
  ) {
    return "glass";
  }

  if (
    value.includes("popier")
  ) {
    return "paper";
  }

  if (
    value.includes("žali") ||
    value.includes("žaliųjų")
  ) {
    return "organic";
  }

  return "mixed";
}

function translateWasteType(
  description
) {
  if (!description) {
    return "Waste collection";
  }

  const value =
    String(
      description
    ).toLowerCase();

  if (
    value.includes("mišri")
  ) {
    return "Mixed waste";
  }

  if (
    value.includes("stiklo")
  ) {
    return "Glass waste";
  }

  if (
    value.includes("popier")
  ) {
    return "Paper & cardboard";
  }

  if (
    value.includes("žali")
  ) {
    return "Green waste";
  }

  return description;
}

function monthName(
  month
) {
  const months = [
    "",
    "January",
    "February",
    "March",
    "April",
    "May",
    "June",
    "July",
    "August",
    "September",
    "October",
    "November",
    "December",
  ];

  return (
    months[month] || ""
  );
}

}

  );
}
